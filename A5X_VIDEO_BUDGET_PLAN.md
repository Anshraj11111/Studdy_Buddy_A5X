#### Create Upload Function:
```javascript
// backend/src/services/bunny.service.js
import axios from 'axios';
import { bunnyConfig } from '../config/bunny.js';

export const uploadToBunny = async (videoFile, title) => {
  // Step 1: Create video entry
  const createResponse = await axios.post(
    `https://video.bunnycdn.com/library/${bunnyConfig.libraryId}/videos`,
    { title },
    {
      headers: {
        'AccessKey': bunnyConfig.apiKey,
        'Content-Type': 'application/json'
      }
    }
  );

  const videoId = createResponse.data.guid;

  // Step 2: Upload video file
  await axios.put(
    `https://video.bunnycdn.com/library/${bunnyConfig.libraryId}/videos/${videoId}`,
    videoFile,
    {
      headers: {
        'AccessKey': bunnyConfig.apiKey,
        'Content-Type': 'application/octet-stream'
      }
    }
  );

  // Step 3: Return video URL
  return {
    videoId,
    playbackUrl: `https://${bunnyConfig.cdnHostname}/${videoId}/playlist.m3u8`,
    thumbnailUrl: `https://${bunnyConfig.cdnHostname}/${videoId}/thumbnail.jpg`
  };
};
```

#### Update Course Controller:
```javascript
// backend/src/controllers/course.controller.js

// Add new upload endpoint
export const uploadLecture = async (req, res) => {
  try {
    const { title, moduleId } = req.body;
    const videoFile = req.file.buffer;

    // Upload to Bunny
    const bunnyData = await uploadToBunny(videoFile, title);

    // Save to database
    const lecture = await Lecture.create({
      title,
      moduleId,
      videoUrl: bunnyData.playbackUrl,
      thumbnailUrl: bunnyData.thumbnailUrl,
      bunnyVideoId: bunnyData.videoId,
      uploadedBy: req.user._id
    });

    res.json({ success: true, lecture });
  } catch (error) {
    res.status(500).json({ error: error.message });
  }
};
```

```jsx
// frontend/src/components/BunnyVideoPlayer.jsx
import { useRef, useEffect } from 'react';
import videojs from 'video.js';
import 'video.js/dist/video-js.css';

function BunnyVideoPlayer({ videoUrl, title, onComplete }) {
  const videoRef = useRef(null);
  const playerRef = useRef(null);

  useEffect(() => {
    if (!playerRef.current && videoRef.current) {
      const player = videojs(videoRef.current, {
        controls: true,
        fluid: true,
        aspectRatio: '16:9',
        playbackRates: [0.5, 0.75, 1, 1.25, 1.5, 2],
        
        controlBar: {
          children: [
            'playToggle',
            'volumePanel',
            'currentTimeDisplay',
            'timeDivider',
            'durationDisplay',
            'progressControl',
            'playbackRateMenuButton',
            'qualitySelector',
            'fullscreenToggle'
          ]
        }
      });

      // Track completion
      player.on('ended', () => {
        console.log('Video completed!');
        onComplete?.();
      });

      // Track progress
      player.on('timeupdate', () => {
        const progress = (player.currentTime() / player.duration()) * 100;
        if (progress % 10 === 0) {
          saveProgress(progress);
        }
      });

      playerRef.current = player;
    }

    return () => {
      if (playerRef.current) {
        playerRef.current.dispose();
        playerRef.current = null;
      }
    };
  }, [videoUrl]);

  return (
    <div className="bunny-video-container">
      <div data-vjs-player>
        <video
          ref={videoRef}
          className="video-js vjs-big-play-centered"
          playsInline
        >
          <source src={videoUrl} type="application/x-mpegURL" />
        </video>
      </div>
      
      <div className="video-branding">
        <span>by A5x</span>
      </div>
    </div>
  );
}

export default BunnyVideoPlayer;
```
